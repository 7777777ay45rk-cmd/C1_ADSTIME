# SEO-NOTE: "?list=..." isn't a URL pattern this site generates anywhere (checked - nothing reads
# $_GET['list'], every page's canonical tag already points at the clean URL regardless of query
# string). These are bot/scraper-guessed URLs, most likely, showing up in Search Console under
# "Alternate page with proper canonical tag" - not an error, just Google correctly excluding the
# duplicate. This line stops Googlebot from spending crawl budget fetching that pattern at all.
# WHY not a blanket "Disallow: /*?*" for every query string instead: real ad-tracking params
# (utm_source, gclid, fbclid, etc.) land on these same pages and must stay crawlable/functional -
# only this specific junk pattern is blocked, nothing else.
Disallow: /*?list=*

# SEO-NOTE: same reasoning as the ?list= rule above - "_bdsid" and "_bd_prev_page" aren't
# parameters this site reads or generates anywhere (this site doesn't even have a detail.php,
# which showed up as the referring page in one example). They're bot/scraper cruft being
# appended to crawled URLs, landing in Search Console as more junk duplicate-URL noise.
# Using "/*_bdsid=*" (no literal "?" right before it) rather than "/*?_bdsid=*" so this still
# matches when the parameter isn't first in the query string, e.g. "...&_bdsid=...".
Disallow: /*_bdsid=*
Disallow: /*_bd_prev_page=*
